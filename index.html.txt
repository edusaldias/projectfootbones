<script>
    // Función para cambiar entre pestañas
    function switchTab(evt, boneName) {
        const sections = document.getElementsByClassName("calculator-section");
        for (let i = 0; i < sections.length; i++) {
            sections[i].classList.remove("active");
        }
        const buttons = document.getElementsByClassName("tab-btn");
        for (let i = 0; i < buttons.length; i++) {
            buttons[i].classList.remove("active");
        }
        document.getElementById(boneName).classList.add("active");
        evt.currentTarget.classList.add("active");
        document.getElementById('result').style.display = "none";
    }

    // Función auxiliar para aplicar la Sigmoidea y mostrar la interfaz
    function displayResult(logit) {
        // P = 1 / (1 + e^(-logit))
        const probability = 1 / (1 + Math.exp(-logit));
        const resultBox = document.getElementById('result');
        const resultTitle = document.getElementById('resultTitle');
        const resultProb = document.getElementById('resultProb');

        resultBox.style.display = "block";

        // Punto de corte estándar (Cut-off = 0.5)
        if (probability >= 0.5) {
            resultBox.className = "result-box male";
            resultTitle.innerText = "Estimated Sex: MALE";
            resultProb.innerText = "Probability: " + (probability * 100).toFixed(2) + "%";
        } else {
            resultBox.className = "result-box female";
            resultTitle.innerText = "Estimated Sex: FEMALE";
            resultProb.innerText = "Probability: " + ((1 - probability) * 100).toFixed(2) + "%";
        }
    }

    // 1. CÁLCULO CUNEIFORME MEDIAL
    function calculateMedial() {
        const mh1 = parseFloat(document.getElementById('mh1').value);
        if (isNaN(mh1)) { alert("Please enter all measurements."); return; }
        
        // Ecuación: -0.1176 + (-2.3256 * MH1)
        const logit = -0.1176 + (-2.3256 * mh1);
        displayResult(logit);
    }

    // 2. CÁLCULO CUNEIFORME INTERMEDIO
    function calculateIntermediate() {
        const ml2 = parseFloat(document.getElementById('ml2').value);
        const mh2 = parseFloat(document.getElementById('mh2').value);
        const lfn2 = parseFloat(document.getElementById('lfn2').value);
        if (isNaN(ml2) || isNaN(mh2) || isNaN(lfn2)) { alert("Please enter all measurements."); return; }

        // Ecuación: -0.08882 + (-0.67539 * ML2) + (-1.13979 * MH2) + (0.52643 * LFN2)
        const logit = -0.08882 + (-0.67539 * ml2) + (-1.13979 * mh2) + (0.52643 * lfn2);
        displayResult(logit);
    }

    // 3. CÁLCULO CUNEIFORME LATERAL
    function calculateLateral() {
        const ml3 = parseFloat(document.getElementById('ml3').value);
        const mh3 = parseFloat(document.getElementById('mh3').value);
        if (isNaN(ml3) || isNaN(mh3)) { alert("Please enter all measurements."); return; }

        // Ecuación: -0.07016 + (-0.67515 * ML3) + (-1.03755 * MH3)
        const logit = -0.07016 + (-0.67515 * ml3) + (-1.03755 * mh3);
        displayResult(logit);
    }
